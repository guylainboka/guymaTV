     1→"use client";
     2→
     3→import React, { useState, useRef, useEffect, useCallback } from "react";
     4→import { StreamItem, StreamServer, StreamEpisode } from "@/lib/types";
     5→import { uBlockEngineInstance } from "@/lib/uBlockEngine";
     6→import { apiUrl } from "@/lib/api-client";
     7→import { UBlockModal } from "./UBlockModal";
     8→
     9→// hls.js is loaded dynamically only when we need to play a .m3u8 stream.
    10→// Native HLS is supported on Safari/iOS, so we only load hls.js for other browsers.
    11→let HlsModule: typeof import("hls.js").default | null = null;
    12→
    13→interface SecureBrowserModalProps {
    14→  item: StreamItem;
    15→  onClose: () => void;
    16→  onStartDownload: (item: StreamItem) => void;
    17→  isFavorite: boolean;
    18→  onToggleFavorite: (id: string) => void;
    19→  onSelectRelatedItem?: (item: StreamItem) => void;
    20→}
    21→
    22→/**
    23→ * Returns true when `url` is a direct video file we can hand to a native
    24→ * `<video>` element (.mp4 / .m3u8 / .webm, or any absolute http(s) URL that
    25→ * isn't our own /api/proxy fallback). Returns false for the proxy URL.
    26→ */
    27→function isDirectVideo(url: string): boolean {
    28→  if (!url) return false;
    29→  if (url.startsWith("/api/proxy")) return false;
    30→  if (/^https?:\/\//i.test(url)) return true;
    31→  return /\.(mp4|m3u8|webm)(\?|$)/i.test(url);
    32→}
    33→
    34→/**
    35→ * Secure in-app stream player modal.
    36→ *
    37→ * Ported from the original maquette with the following adjustments:
    38→ *  - The fake URL/address bar (lock icon, domain, path, "SSL Sécurisé" badge)
    39→ *    is REMOVED entirely so only the video and player controls remain.
    40→ *  - On mount we fetch the real extracted servers from
    41→ *    `GET /api/extract?id=${item.id}` (Playwright, cached 30 min server-side).
    42→ *    A spinner with "Extraction des lecteurs en cours…" is shown while
    43→ *    Playwright runs (~10-15s on first call).
    44→ *  - Smart rendering: if `currentServer.videoUrl` is a direct video URL
    45→ *    (mp4/m3u8/webm or any http(s) URL that isn't /api/proxy), we render a
    46→ *    native `<video>` element and wire every player control to it via refs
    47→ *    (play/pause, timeline scrub, volume, skip ±10s, fullscreen). If the URL
    48→ *    is `/api/proxy?page=...` (extraction failed/empty), we fall back to the
    49→ *    filtered `<iframe>` as before.
    50→ *  - The "Site Complet (Nettoyé)" view mode toggle is removed — only the
    51→ *    cinematic player view remains.
    52→ *  - The maquette chrome (top action bar, uBlock badge, player stage, bottom
    53→ *    control bar, server selector, episode selector, media details, uBlock
    54→ *    protection banner) is preserved identically.
    55→ */
    56→export const SecureBrowserModal: React.FC<SecureBrowserModalProps> = ({
    57→  item,
    58→  onClose,
    59→  onStartDownload,
    60→  isFavorite,
    61→  onToggleFavorite,
    62→  onSelectRelatedItem,
    63→}) => {
    64→  // ---- Server fetch state (populated from /api/extract on mount) ----
    65→  const [servers, setServers] = useState<StreamServer[]>([]);
    66→  const [serversLoading, setServersLoading] = useState(true);
    67→
    68→  // ---- Player state ----
    69→  const [selectedServerIndex, setSelectedServerIndex] = useState(0);
    70→  const [selectedEpisodeIndex, setSelectedEpisodeIndex] = useState(0);
    71→  const [isPlaying, setIsPlaying] = useState(true);
    72→  const [isFullscreen, setIsFullscreen] = useState(false);
    73→  const [currentTime, setCurrentTime] = useState(0);
    74→  const [duration, setDuration] = useState(0);
    75→  const [volume, setVolume] = useState(85);
    76→  const [isMuted, setIsMuted] = useState(false);
    77→  const [showUBlockModal, setShowUBlockModal] = useState(false);
    78→  const [blockedAdsOnPage, setBlockedAdsOnPage] = useState(
    79→    (item.blockedAdStats?.popups || 12) + (item.blockedAdStats?.adultBanners || 6)
    80→  );
    81→  const [iframeKey, setIframeKey] = useState(0);
    82→  const [switchingServer, setSwitchingServer] = useState(false);
    83→
    84→  const containerRef = useRef<HTMLDivElement>(null);
    85→  const videoRef = useRef<HTMLVideoElement>(null);
    86→
    87→  // Build the proxy fallback server (used when extraction fails or returns []).
    88→  const buildProxyFallback = useCallback((): StreamServer => ({
    89→    id: `srv-${item.id}-secure`,
    90→    name: "Lecteur Sécurisé (proxy)",
    91→    hoster: "Direct 4K",
    92→    quality: item.quality,
    93→    language: "VF",
    94→    speed: "Ultra Rapide (sans pub)",
    95→    videoUrl: apiUrl(`/api/proxy?page=${encodeURIComponent(item.id)}`),
    96→  }), [item.id, item.quality]);
    97→
    98→  // Fetch real extracted servers from /api/extract on mount (and when item.id changes).
    99→  useEffect(() => {
   100→    let cancelled = false;
   101→    setServersLoading(true);
   102→    setSwitchingServer(false);
   103→    setSelectedServerIndex(0);
   104→    setSelectedEpisodeIndex(0);
   105→    setCurrentTime(0);
   106→    setDuration(0);
   107→    setIsPlaying(true);
   108→
   109→    fetch(apiUrl(`/api/extract?id=${encodeURIComponent(item.id)}`))
   110→      .then((r) => {
   111→        if (!r.ok) throw new Error(`extract failed: ${r.status}`);
   112→        return r.json();
   113→      })
   114→      .then((data: StreamServer[] | { servers?: StreamServer[] }) => {
   115→        if (cancelled) return;
   116→        const list: StreamServer[] = Array.isArray(data)
   117→          ? data
   118→          : (data?.servers || []);
   119→        if (list.length > 0) {
   120→          setServers(list);
   121→        } else {
   122→          // Empty result — fall back to a single proxy server.
   123→          setServers([buildProxyFallback()]);
   124→        }
   125→      })
   126→      .catch(() => {
   127→        if (cancelled) return;
   128→        // Network/extractor error — fall back to a single proxy server.
   129→        setServers([buildProxyFallback()]);
   130→      })
   131→      .finally(() => {
   132→        if (!cancelled) setServersLoading(false);
   133→      });
   134→
   135→    return () => { cancelled = true; };
   136→  }, [item.id, buildProxyFallback]);
   137→
   138→  const currentServer = servers[selectedServerIndex] || servers[0];
   139→  const episodes: StreamEpisode[] = item.episodes || [];
   140→  const currentEpisode = episodes[selectedEpisodeIndex];
   141→
   142→  // Resolve the URL to actually play. Episode-specific videoUrl wins for
   143→  // series, otherwise the current server's videoUrl, otherwise the proxy.
   144→  const currentVideoUrl: string =
   145→    currentEpisode?.videoUrl ||
   146→    currentServer?.videoUrl ||
   147→    apiUrl(`/api/proxy?page=${encodeURIComponent(item.id)}`);
   148→
   149→  const directVideo = isDirectVideo(currentVideoUrl);
   150→  const isHlsStream = directVideo && /\.m3u8(\?|$)/i.test(currentVideoUrl);
   151→
   152→  // Attach HLS.js to the <video> element when the stream is a .m3u8 URL.
   153→  // Native HLS (Safari/iOS) doesn't need hls.js — it plays .m3u8 directly.
   154→  useEffect(() => {
   155→    if (!directVideo || !isHlsStream) return;
   156→    const video = videoRef.current;
   157→    if (!video) return;
   158→
   159→    let hls: import("hls.js").default | null = null;
   160→
   161→    // Native HLS support (Safari, iOS)
   162→    if (video.canPlayType("application/vnd.apple.mpegurl")) {
   163→      video.src = currentVideoUrl;
   164→      video.play().catch(() => {});
   165→      return () => {
   166→        video.removeAttribute("src");
   167→        video.load();
   168→      };
   169→    }
   170→
   171→    // Other browsers → use hls.js
   172→    let cancelled = false;
   173→    (async () => {
   174→      if (!HlsModule) {
   175→        HlsModule = (await import("hls.js")).default;
   176→      }
   177→      if (cancelled || !HlsModule) return;
   178→      if (!HlsModule.isSupported()) return;
   179→      hls = new HlsModule({ enableWorker: true, lowLatencyMode: false });
   180→      hls.loadSource(currentVideoUrl);
   181→      hls.attachMedia(video);
   182→      hls.on(HlsModule.Events.MANIFEST_PARSED, () => {
   183→        video.play().catch(() => {});
   184→        setSwitchingServer(false);
   185→      });
   186→      hls.on(HlsModule.Events.ERROR, (_evt, data) => {
   187→        if (data.fatal) {
   188→          console.error("[hls.js] fatal error:", data);
   189→          setSwitchingServer(false);
   190→        }
   191→      });
   192→    })();
   193→
   194→    return () => {
   195→      cancelled = true;
   196→      if (hls) {
   197→        hls.destroy();
   198→      }
   199→    };
   200→  }, [currentVideoUrl, directVideo, isHlsStream]);
   201→
   202→  const domain = item.sourceServiceId
   203→    ? `${item.sourceServiceId}.net`
   204→    : "french-stream.net";
   205→
   206→  // =========================================================================
   207→  // Native <video> controls — every control below is wired to videoRef.
   208→  // =========================================================================
   209→
   210→  const togglePlay = useCallback(() => {
   211→    const v = videoRef.current;
   212→    if (!v) {
   213→      setIsPlaying((prev) => !prev);
   214→      return;
   215→    }
   216→    if (v.paused) {
   217→      v.play().then(() => setIsPlaying(true)).catch(() => {});
   218→    } else {
   219→      v.pause();
   220→      setIsPlaying(false);
   221→    }
   222→  }, []);
   223→
   224→  const toggleFullscreen = useCallback(() => {
   225→    if (!containerRef.current) return;
   226→    if (!document.fullscreenElement) {
   227→      containerRef.current
   228→        .requestFullscreen?.()
   229→        .then(() => setIsFullscreen(true))
   230→        .catch(() => {});
   231→    } else {
   232→      document.exitFullscreen?.()
   233→        .then(() => setIsFullscreen(false))
   234→        .catch(() => {});
   235→    }
   236→  }, []);
   237→
   238→  const skipBy = useCallback((delta: number) => {
   239→    const v = videoRef.current;
   240→    if (!v) return;
   241→    const target = Math.max(0, Math.min(v.duration || 0, v.currentTime + delta));
   242→    v.currentTime = target;
   243→    setCurrentTime(target);
   244→  }, []);
   245→
   246→  const handleSeek = (e: React.MouseEvent<HTMLDivElement>) => {
   247→    const v = videoRef.current;
   248→    if (!v || !v.duration) return;
   249→    const rect = e.currentTarget.getBoundingClientRect();
   250→    const pos = (e.clientX - rect.left) / rect.width;
   251→    const newTime = Math.max(0, Math.min(v.duration, pos * v.duration));
   252→    v.currentTime = newTime;
   253→    setCurrentTime(newTime);
   254→  };
   255→
   256→  const handleVolumeChange = (e: React.ChangeEvent<HTMLInputElement>) => {
   257→    const newVol = Number(e.target.value);
   258→    setVolume(newVol);
   259→    setIsMuted(false);
   260→    const v = videoRef.current;
   261→    if (v) {
   262→      v.volume = newVol / 100;
   263→      v.muted = false;
   264→    }
   265→  };
   266→
   267→  const toggleMute = useCallback(() => {
   268→    const v = videoRef.current;
   269→    if (!v) {
   270→      setIsMuted((p) => !p);
   271→      return;
   272→    }
   273→    v.muted = !v.muted;
   274→  }, []);
   275→
   276→  const formatSeconds = (sec: number) => {
   277→    if (!sec || !isFinite(sec)) return "0:00";
   278→    const mins = Math.floor(sec / 60);
   279→    const s = Math.floor(sec % 60);
   280→    return `${mins}:${s < 10 ? "0" : ""}${s}`;
   281→  };
   282→
   283→  const handleServerSwitch = (idx: number) => {
   284→    if (idx === selectedServerIndex) return;
   285→    setSwitchingServer(true);
   286→    setSelectedServerIndex(idx);
   287→    setSelectedEpisodeIndex(0);
   288→    setIsPlaying(true);
   289→    setCurrentTime(0);
   290→    setDuration(0);
   291→  };
   292→
   293→  const handleEpisodeSwitch = (idx: number) => {
   294→    if (idx === selectedEpisodeIndex) return;
   295→    setSwitchingServer(true);
   296→    setSelectedEpisodeIndex(idx);
   297→    setIsPlaying(true);
   298→    setCurrentTime(0);
   299→    setDuration(0);
   300→  };
   301→
   302→  const handleRefresh = () => {
   303→    // Re-filter ads and reload the video/iframe.
   304→    uBlockEngineInstance.inspectUrl("https://syndication.exoclick.com/splash.php?partner=french-stream");
   305→    uBlockEngineInstance.inspectUrl("https://serve.popads.net/serve.js?partner=french-stream");
   306→    setBlockedAdsOnPage((prev) => prev + 2);
   307→    setIframeKey((k) => k + 1);
   308→    setSwitchingServer(true);
   309→    // For native video, force a reload by re-assigning src and calling load().
   310→    if (videoRef.current && directVideo) {
   311→      const v = videoRef.current;
   312→      const src = v.src;
   313→      v.removeAttribute("src");
   314→      v.load();
   315→      v.src = src;
   316→      v.load();
   317→      v.play().then(() => setIsPlaying(true)).catch(() => {});
   318→    }
   319→  };
   320→
   321→  // Track fullscreen changes (so Esc from fullscreen updates the icon).
   322→  useEffect(() => {
   323→    const onFsChange = () => setIsFullscreen(!!document.fullscreenElement);
   324→    document.addEventListener("fullscreenchange", onFsChange);
   325→    return () => document.removeEventListener("fullscreenchange", onFsChange);
   326→  }, []);
   327→
   328→  // Keyboard navigation & remote-control shortcuts.
   329→  useEffect(() => {
   330→    const handleKeyDown = (e: KeyboardEvent) => {
   331→      if (["INPUT", "TEXTAREA"].includes((e.target as HTMLElement)?.tagName)) return;
   332→
   333→      if (e.key === "Escape") {
   334→        // If we're in fullscreen, let the browser exit fullscreen first;
   335→        // only close the modal when not in fullscreen.
   336→        if (!document.fullscreenElement) {
   337→          e.preventDefault();
   338→          onClose();
   339→        }
   340→      } else if (e.key === " " || e.key === "Enter") {
   341→        e.preventDefault();
   342→        togglePlay();
   343→      } else if (e.key === "f" || e.key === "F") {
   344→        e.preventDefault();
   345→        toggleFullscreen();
   346→      } else if (e.key === "ArrowUp") {
   347→        e.preventDefault();
   348→        const newVol = Math.min(100, volume + 10);
   349→        setVolume(newVol);
   350→        setIsMuted(false);
   351→        if (videoRef.current) {
   352→          videoRef.current.volume = newVol / 100;
   353→          videoRef.current.muted = false;
   354→        }
   355→      } else if (e.key === "ArrowDown") {
   356→        e.preventDefault();
   357→        const newVol = Math.max(0, volume - 10);
   358→        setVolume(newVol);
   359→        if (videoRef.current) {
   360→          videoRef.current.volume = newVol / 100;
   361→        }
   362→      } else if (e.key === "m" || e.key === "M") {
   363→        e.preventDefault();
   364→        toggleMute();
   365→      } else if (e.key === "ArrowLeft") {
   366→        e.preventDefault();
   367→        skipBy(-10);
   368→      } else if (e.key === "ArrowRight") {
   369→        e.preventDefault();
   370→        skipBy(10);
   371→      }
   372→    };
   373→
   374→    window.addEventListener("keydown", handleKeyDown);
   375→    return () => window.removeEventListener("keydown", handleKeyDown);
   376→  }, [volume, togglePlay, toggleFullscreen, toggleMute, skipBy, onClose]);
   377→
   378→  // Server label: proxy fallbacks get the "Lecteur Sécurisé (proxy)" label,
   379→  // direct-video servers show their hoster name.
   380→  const serverLabel = (srv: StreamServer): string => {
   381→    if (srv.videoUrl && !isDirectVideo(srv.videoUrl)) {
   382→      return "Lecteur Sécurisé (proxy)";
   383→    }
   384→    return srv.name || srv.hoster;
   385→  };
   386→
   387→  // Timeline fill percentage (guard against div-by-zero before metadata loads).
   388→  const timelinePct = duration > 0 ? (currentTime / duration) * 100 : 0;
   389→
   390→  return (
   391→    <div
   392→      className="fixed inset-0 z-50 flex items-center justify-center p-2 sm:p-4 bg-black/90 backdrop-blur-md animate-fade-in"
   393→      onClick={onClose}
   394→    >
   395→      <div
   396→        ref={containerRef}
   397→        className="w-full max-w-6xl bg-surface-container-high rounded-2xl overflow-hidden shadow-2xl border border-outline-variant/20 flex flex-col h-[94vh]"
   398→        onClick={(e) => e.stopPropagation()}
   399→      >
   400→        {/* Secure In-App Browser Top Action Bar (URL bar removed) */}
   401→        <div className="bg-surface-container-highest px-3 py-2 border-b border-outline-variant/15 flex items-center justify-between gap-2 flex-wrap">
   402→          {/* Left: Navigation Controls (back & refresh) + uBlock blocked-ads pill */}
   403→          <div className="flex items-center gap-1 text-on-surface-variant shrink-0">
   404→            <button
   405→              onClick={onClose}
   406→              className="w-8 h-8 rounded-full hover:bg-surface-container flex items-center justify-center transition-colors cursor-pointer"
   407→              title="Fermer (Échap)"
   408→            >
   409→              <span className="material-symbols-outlined text-[18px]">arrow_back</span>
   410→            </button>
   411→            <button
   412→              onClick={handleRefresh}
   413→              className="w-8 h-8 rounded-full hover:bg-surface-container flex items-center justify-center transition-colors cursor-pointer"
   414→              title="Actualiser et re-filtrer les pubs"
   415→            >
   416→              <span className="material-symbols-outlined text-[18px]">refresh</span>
   417→            </button>
   418→            <div className="flex items-center gap-1.5 ml-2 px-2.5 py-1 rounded-full bg-primary/10 border border-primary/30 text-primary text-[11px] font-semibold">
   419→              <span className="material-symbols-outlined text-[14px]">shield</span>
   420→              <span className="font-bold">{blockedAdsOnPage} pubs bloquées</span>
   421→              <span className="w-1.5 h-1.5 rounded-full bg-primary animate-pulse" />
   422→            </div>
   423→          </div>
   424→
   425→          {/* Right: Action Buttons (uBlock, download, favorite, fullscreen, close) */}
   426→          <div className="flex items-center gap-2">
   427→            {/* uBlock Origin Shield Trigger */}
   428→            <button
   429→              onClick={() => setShowUBlockModal(true)}
   430→              className="flex items-center gap-1.5 bg-primary/10 hover:bg-primary/20 text-primary text-[11px] font-semibold px-2.5 py-1.5 rounded-full border border-primary/30 transition-all cursor-pointer"
   431→              title="Ouvrir le panneau uBlock Origin"
   432→            >
   433→              <span className="material-symbols-outlined text-[15px]">shield</span>
   434→              <span className="font-bold hidden md:inline">uBlock</span>
   435→              <span className="w-1.5 h-1.5 rounded-full bg-primary animate-pulse" />
   436→            </button>
   437→
   438→            {/* Direct Download button */}
   439→            <button
   440→              onClick={() => onStartDownload(item)}
   441→              className="px-3 py-1.5 rounded-full bg-primary text-on-primary text-label-sm font-bold flex items-center gap-1 hover:scale-105 transition-all cursor-pointer shadow-sm"
   442→              title="Télécharger directement la vidéo sans pub"
   443→            >
   444→              <span className="material-symbols-outlined text-[16px]">download</span>
   445→              <span className="hidden md:inline">Télécharger</span>
   446→            </button>
   447→
   448→            {/* Favorite toggle */}
   449→            <button
   450→              onClick={() => onToggleFavorite(item.id)}
   451→              className="w-8 h-8 rounded-full bg-surface-container flex items-center justify-center text-on-surface hover:text-primary transition-colors cursor-pointer"
   452→              title={isFavorite ? "Retirer des favoris" : "Ajouter aux favoris"}
   453→            >
   454→              <span
   455→                className={`material-symbols-outlined text-[18px] ${
   456→                  isFavorite ? "text-primary fill-current" : ""
   457→                }`}
   458→                style={isFavorite ? { fontVariationSettings: "'FILL' 1" } : undefined}
   459→              >
   460→                favorite
   461→              </span>
   462→            </button>
   463→
   464→            {/* Fullscreen toggle */}
   465→            <button
   466→              onClick={toggleFullscreen}
   467→              className="w-8 h-8 rounded-full bg-surface-container flex items-center justify-center text-on-surface hover:text-primary transition-colors cursor-pointer"
   468→              title="Plein écran (F)"
   469→            >
   470→              <span className="material-symbols-outlined text-[18px]">
   471→                {isFullscreen ? "fullscreen_exit" : "fullscreen"}
   472→              </span>
   473→            </button>
   474→
   475→            {/* Close */}
   476→            <button
   477→              onClick={onClose}
   478→              className="w-8 h-8 rounded-full bg-surface-container flex items-center justify-center text-on-surface-variant hover:text-on-surface transition-colors cursor-pointer"
   479→              title="Fermer"
   480→            >
   481→              <span className="material-symbols-outlined text-[18px]">close</span>
   482→            </button>
   483→          </div>
   484→        </div>
   485→
   486→        {/* Modal Body — Player view only (Site Complet view removed) */}
   487→        <div className="flex-1 overflow-y-auto flex flex-col bg-surface-container-low">
   488→          <div className="flex flex-col flex-1">
   489→            {/* Top Video Player Stage — native <video> or filtered <iframe> */}
   490→            <div className="relative aspect-video w-full bg-black flex items-center justify-center overflow-hidden group select-none max-h-[58vh]">
   491→              {serversLoading ? (
   492→                /* ---- Extraction loading state ---- */
   493→                <div className="absolute inset-0 flex flex-col items-center justify-center gap-4 bg-surface-container-low">
   494→                  <div className="w-12 h-12 border-4 border-primary/30 border-t-primary rounded-full animate-spin" />
   495→                  <div className="text-center px-4">
   496→                    <p className="text-on-surface font-semibold text-body-md">
   497→                      Extraction des lecteurs en cours…
   498→                    </p>
   499→                    <p className="text-on-surface-variant text-[12px] mt-1">
   500→                      Analyse de la page source via Playwright (~10-15s au premier appel)
   501→                    </p>
   502→                  </div>
   503→                </div>
   504→              ) : directVideo ? (
   505→                /* ---- Native <video> player (direct mp4/m3u8/webm URL) ---- */
   506→                <video
   507→                  ref={videoRef}
   508→                  key={`video-${iframeKey}-${selectedServerIndex}-${selectedEpisodeIndex}`}
   509→                  src={isHlsStream ? undefined : currentVideoUrl}
   510→                  autoPlay
   511→                  playsInline
   512→                  controls={false}
   513→                  className="w-full h-full object-contain bg-black"
   514→                  onPlay={() => setIsPlaying(true)}
   515→                  onPause={() => setIsPlaying(false)}
