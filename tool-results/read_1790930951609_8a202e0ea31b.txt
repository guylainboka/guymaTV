    13→    13→interface SecureBrowserModalProps {
    14→    14→  item: StreamItem;
    15→    15→  onClose: () => void;
    16→    16→  onStartDownload: (item: StreamItem) => void;
    17→    17→  isFavorite: boolean;
    18→    18→  onToggleFavorite: (id: string) => void;
    19→    19→  onSelectRelatedItem?: (item: StreamItem) => void;

    21→    21→
    22→    22→/**
    23→    23→ * Returns true when `url` is a direct video file we can hand to a native
    24→    24→ * `<video>` element (.mp4 / .m3u8 / .webm, or any absolute http(s) URL that
    25→    25→ * isn't our own /api/proxy fallback). Returns false for the proxy URL.
    26→    26→ */
    27→    27→function isDirectVideo(url: string): boolean {
    28→    28→  if (!url) return false;
    29→    29→  if (url.startsWith("/api/proxy")) return false;
    30→    30→  if (/^https?:\/\//i.test(url)) return true;
    31→    31→  return /\.(mp4|m3u8|webm)(\?|$)/i.test(url);
    32→    32→}
    33→    33→
    34→    34→/**
    35→    35→ * Secure in-app stream player modal.
    36→    36→ *
    37→    37→ * Ported from the original maquette with the following adjustments:
    38→    38→ *  - The fake URL/address bar (lock icon, domain, path, "SSL Sécurisé" badge)
    39→    39→ *    is REMOVED entirely so only the video and player controls remain.
    40→    40→ *  - On mount we fetch the real extracted servers from
    41→    41→ *    `GET /api/extract?id=${item.id}` (Playwright, cached 30 min server-side).
    42→    42→ *    A spinner with "Extraction des lecteurs en cours…" is shown while
    43→    43→ *    Playwright runs (~10-15s on first call).
    44→    44→ *  - Smart rendering: if `currentServer.videoUrl` is a direct video URL
    45→    45→ *    (mp4/m3u8/webm or any http(s) URL that isn't /api/proxy), we render a
    46→    46→ *    native `<video>` element and wire every player control to it via refs
    47→    47→ *    (play/pause, timeline scrub, volume, skip ±10s, fullscreen). If the URL
    48→    48→ *    is `/api/proxy?page=...` (extraction failed/empty), we fall back to the
    49→    49→ *    filtered `<iframe>` as before.
    50→    50→ *  - The "Site Complet (Nettoyé)" view mode toggle is removed — only the
    51→    51→ *    cinematic player view remains.
    52→    52→ *  - The maquette chrome (top action bar, uBlock badge, player stage, bottom
    53→    53→ *    control bar, server selector, episode selector, media details, uBlock
    54→    54→ *    protection banner) is preserved identically.
    55→    55→ */
    56→    56→export const SecureBrowserModal: React.FC<SecureBrowserModalProps> = ({
    57→    57→  item,
    58→    58→  onClose,
    59→    59→  onStartDownload,
    60→    60→  isFavorite,
    61→    61→  onToggleFavorite,
    62→    62→  onSelectRelatedItem,
    63→    63→}) => {
    64→    64→  // ---- Server fetch state (populated from /api/extract on mount) ----
    65→    65→  const [servers, setServers] = useState<StreamServer[]>([]);
    66→    66→  const [serversLoading, setServersLoading] = useState(true);
    67→    67→
    68→    68→  // ---- Player state ----
    69→    69→  const [selectedServerIndex, setSelectedServerIndex] = useState(0);
    70→    70→  const [selectedEpisodeIndex, setSelectedEpisodeIndex] = useState(0);
    71→    71→  const [isPlaying, setIsPlaying] = useState(true);
    72→    72→  const [isFullscreen, setIsFullscreen] = useState(false);
    73→    73→  const [currentTime, setCurrentTime] = useState(0);
    74→    74→  const [duration, setDuration] = useState(0);
    75→    75→  const [volume, setVolume] = useState(85);
    76→    76→  const [isMuted, setIsMuted] = useState(false);
    77→    77→  const [showUBlockModal, setShowUBlockModal] = useState(false);
    78→    78→  const [blockedAdsOnPage, setBlockedAdsOnPage] = useState(
    79→    79→    (item.blockedAdStats?.popups || 12) + (item.blockedAdStats?.adultBanners || 6)
    80→    80→  );
    81→    81→  const [iframeKey, setIframeKey] = useState(0);
    82→    82→  const [switchingServer, setSwitchingServer] = useState(false);
    83→    83→
    84→    84→  const containerRef = useRef<HTMLDivElement>(null);
    85→    85→  const videoRef = useRef<HTMLVideoElement>(null);
    86→    86→
    87→    87→  // Build the proxy fallback server (used when extraction fails or returns []).
    88→    88→  const buildProxyFallback = useCallback((): StreamServer => ({
    89→    89→    id: `srv-${item.id}-secure`,
    90→    90→    name: "Lecteur Sécurisé (proxy)",
    91→    91→    hoster: "Direct 4K",
    92→    92→    quality: item.quality,
    93→    93→    language: "VF",
    94→    94→    speed: "Ultra Rapide (sans pub)",
    95→    95→    videoUrl: apiUrl(`/api/proxy?page=${encodeURIComponent(item.id)}`),
    96→    96→  }), [item.id, item.quality]);
    97→    97→
    98→    98→  // Fetch real extracted servers from /api/extract on mount (and when item.id changes).
    99→    99→  useEffect(() => {
   100→   100→    let cancelled = false;
   101→   101→    setServersLoading(true);
   102→   102→    setSwitchingServer(false);
   103→   103→    setSelectedServerIndex(0);
   104→   104→    setSelectedEpisodeIndex(0);
   105→   105→    setCurrentTime(0);
   106→   106→    setDuration(0);
   107→   107→    setIsPlaying(true);
   108→   108→
   109→   109→    fetch(apiUrl(`/api/extract?id=${encodeURIComponent(item.id)}`))
   110→   110→      .then((r) => {
   111→   111→        if (!r.ok) throw new Error(`extract failed: ${r.status}`);
   112→   112→        return r.json();
   113→   113→      })
   114→   114→      .then((data: StreamServer[] | { servers?: StreamServer[] }) => {
   115→   115→        if (cancelled) return;
   116→   116→        const list: StreamServer[] = Array.isArray(data)
   117→   117→          ? data
   118→   118→          : (data?.servers || []);
   119→   119→        if (list.length > 0) {
   120→   120→          setServers(list);
   121→   121→        } else {
   122→   122→          // Empty result — fall back to a single proxy server.
   123→   123→          setServers([buildProxyFallback()]);
   124→   124→        }
   125→   125→      })
   126→   126→      .catch(() => {
   127→   127→        if (cancelled) return;
   128→   128→        // Network/extractor error — fall back to a single proxy server.
   129→   129→        setServers([buildProxyFallback()]);
   130→   130→      })
   131→   131→      .finally(() => {
   132→   132→        if (!cancelled) setServersLoading(false);
   133→   133→      });
   134→   134→
   135→   135→    return () => { cancelled = true; };
   136→   136→  }, [item.id, buildProxyFallback]);
   137→   137→
   138→   138→  const currentServer = servers[selectedServerIndex] || servers[0];
   139→   139→  const episodes: StreamEpisode[] = item.episodes || [];
   140→   140→  const currentEpisode = episodes[selectedEpisodeIndex];
   141→   141→
   142→   142→  // Resolve the URL to actually play. Episode-specific videoUrl wins for
   143→   143→  // series, otherwise the current server's videoUrl, otherwise the proxy.
   144→   144→  const currentVideoUrl: string =
   145→   145→    currentEpisode?.videoUrl ||
   146→   146→    currentServer?.videoUrl ||
   147→   147→    apiUrl(`/api/proxy?page=${encodeURIComponent(item.id)}`);
   148→   148→
   149→   149→  const directVideo = isDirectVideo(currentVideoUrl);
   150→   150→  const isHlsStream = directVideo && /\.m3u8(\?|$)/i.test(currentVideoUrl);
   151→   151→
   152→   152→  // Attach HLS.js to the <video> element when the stream is a .m3u8 URL.
   153→   153→  // Native HLS (Safari/iOS) doesn't need hls.js — it plays .m3u8 directly.
   154→   154→  useEffect(() => {
   155→   155→    if (!directVideo || !isHlsStream) return;
   156→   156→    const video = videoRef.current;
   157→   157→    if (!video) return;
   158→   158→
   159→   159→    let hls: import("hls.js").default | null = null;
   160→   160→
   161→   161→    // Native HLS support (Safari, iOS)
   162→   162→    if (video.canPlayType("application/vnd.apple.mpegurl")) {
   163→   163→      video.src = currentVideoUrl;
   164→   164→      video.play().catch(() => {});
   165→   165→      return () => {
   166→   166→        video.removeAttribute("src");
   167→   167→        video.load();
   168→   168→      };
   169→   169→    }
   170→   170→
   171→   171→    // Other browsers → use hls.js
   172→   172→    let cancelled = false;
   173→   173→    (async () => {
   174→   174→      if (!HlsModule) {
   175→   175→        HlsModule = (await import("hls.js")).default;
   176→   176→      }
   177→   177→      if (cancelled || !HlsModule) return;
   178→   178→      if (!HlsModule.isSupported()) return;
   179→   179→      hls = new HlsModule({ enableWorker: true, lowLatencyMode: false });
   180→   180→      hls.loadSource(currentVideoUrl);
   181→   181→      hls.attachMedia(video);
   182→   182→      hls.on(HlsModule.Events.MANIFEST_PARSED, () => {
   183→   183→        video.play().catch(() => {});
   184→   184→        setSwitchingServer(false);
   185→   185→      });
   186→   186→      hls.on(HlsModule.Events.ERROR, (_evt, data) => {
   187→   187→        if (data.fatal) {
   188→   188→          console.error("[hls.js] fatal error:", data);
   189→   189→          setSwitchingServer(false);
   190→   190→        }
   191→   191→      });
   192→   192→    })();
   193→   193→
   194→   194→    return () => {
   195→   195→      cancelled = true;
   196→   196→      if (hls) {
   197→   197→        hls.destroy();
   198→   198→      }
   199→   199→    };
   200→   200→  }, [currentVideoUrl, directVideo, isHlsStream]);
   201→   201→
   202→   202→  const domain = item.sourceServiceId
   203→   203→    ? `${item.sourceServiceId}.net`
   204→   204→    : "french-stream.net";
   205→   205→
   206→   206→  // =========================================================================
   207→   207→  // Native <video> controls — every control below is wired to videoRef.
   208→   208→  // =========================================================================
   209→   209→
   210→   210→  const togglePlay = useCallback(() => {
   211→   211→    const v = videoRef.current;
   212→   212→    if (!v) {
   213→   213→      setIsPlaying((prev) => !prev);
   214→   214→      return;
   215→   215→    }
   216→   216→    if (v.paused) {
   217→   217→      v.play().then(() => setIsPlaying(true)).catch(() => {});
   218→   218→    } else {
   219→   219→      v.pause();
   220→   220→      setIsPlaying(false);
   221→   221→    }
   222→   222→  }, []);
   223→   223→
   224→   224→  const toggleFullscreen = useCallback(() => {
   225→   225→    if (!containerRef.current) return;
   226→   226→    if (!document.fullscreenElement) {
   227→   227→      containerRef.current
   228→   228→        .requestFullscreen?.()
   229→   229→        .then(() => setIsFullscreen(true))
   230→   230→        .catch(() => {});
   231→   231→    } else {
   232→   232→      document.exitFullscreen?.()
   233→   233→        .then(() => setIsFullscreen(false))
   234→   234→        .catch(() => {});
   235→   235→    }
   236→   236→  }, []);
   237→   237→
   238→   238→  const skipBy = useCallback((delta: number) => {
   239→   239→    const v = videoRef.current;
   240→   240→    if (!v) return;
   241→   241→    const target = Math.max(0, Math.min(v.duration || 0, v.currentTime + delta));
   242→   242→    v.currentTime = target;
   243→   243→    setCurrentTime(target);
   244→   244→  }, []);
   245→   245→
   246→   246→  const handleSeek = (e: React.MouseEvent<HTMLDivElement>) => {
   247→   247→    const v = videoRef.current;
   248→   248→    if (!v || !v.duration) return;
   249→   249→    const rect = e.currentTarget.getBoundingClientRect();
   250→   250→    const pos = (e.clientX - rect.left) / rect.width;
   251→   251→    const newTime = Math.max(0, Math.min(v.duration, pos * v.duration));
   252→   252→    v.currentTime = newTime;
   253→   253→    setCurrentTime(newTime);
   254→   254→  };
   255→   255→
   256→   256→  const handleVolumeChange = (e: React.ChangeEvent<HTMLInputElement>) => {
   257→   257→    const newVol = Number(e.target.value);
   258→   258→    setVolume(newVol);
   259→   259→    setIsMuted(false);
   260→   260→    const v = videoRef.current;
   261→   261→    if (v) {
   262→   262→      v.volume = newVol / 100;
   263→   263→      v.muted = false;
   264→   264→    }
   265→   265→  };
   266→   266→
   267→   267→  const toggleMute = useCallback(() => {
   268→   268→    const v = videoRef.current;
   269→   269→    if (!v) {
   270→   270→      setIsMuted((p) => !p);
   271→   271→      return;
   272→   272→    }
   273→   273→    v.muted = !v.muted;
   274→   274→  }, []);
   275→   275→
   276→   276→  const formatSeconds = (sec: number) => {
   277→   277→    if (!sec || !isFinite(sec)) return "0:00";
   278→   278→    const mins = Math.floor(sec / 60);
   279→   279→    const s = Math.floor(sec % 60);
   280→   280→    return `${mins}:${s < 10 ? "0" : ""}${s}`;
   281→   281→  };
   282→   282→
   283→   283→  const handleServerSwitch = (idx: number) => {
   284→   284→    if (idx === selectedServerIndex) return;
   285→   285→    setSwitchingServer(true);
   286→   286→    setSelectedServerIndex(idx);
   287→   287→    setSelectedEpisodeIndex(0);
   288→   288→    setIsPlaying(true);
   289→   289→    setCurrentTime(0);
   290→   290→    setDuration(0);
   291→   291→  };
   292→   292→
   293→   293→  const handleEpisodeSwitch = (idx: number) => {
   294→   294→    if (idx === selectedEpisodeIndex) return;
   295→   295→    setSwitchingServer(true);
   296→   296→    setSelectedEpisodeIndex(idx);
   297→   297→    setIsPlaying(true);
   298→   298→    setCurrentTime(0);
   299→   299→    setDuration(0);
   300→   300→  };
   301→   301→
   302→   302→  const handleRefresh = () => {
   303→   303→    // Re-filter ads and reload the video/iframe.
   304→   304→    uBlockEngineInstance.inspectUrl("https://syndication.exoclick.com/splash.php?partner=french-stream");
   305→   305→    uBlockEngineInstance.inspectUrl("https://serve.popads.net/serve.js?partner=french-stream");
   306→   306→    setBlockedAdsOnPage((prev) => prev + 2);
   307→   307→    setIframeKey((k) => k + 1);
   308→   308→    setSwitchingServer(true);
   309→   309→    // For native video, force a reload by re-assigning src and calling load().
   310→   310→    if (videoRef.current && directVideo) {
   311→   311→      const v = videoRef.current;
   312→   312→      const src = v.src;
   313→   313→      v.removeAttribute("src");
   314→   314→      v.load();
   315→   315→      v.src = src;
   316→   316→      v.load();
   317→   317→      v.play().then(() => setIsPlaying(true)).catch(() => {});
   318→   318→    }
   319→   319→  };
   320→   320→
   321→   321→  // Track fullscreen changes (so Esc from fullscreen updates the icon).
   322→   322→  useEffect(() => {
   323→   323→    const onFsChange = () => setIsFullscreen(!!document.fullscreenElement);
   324→   324→    document.addEventListener("fullscreenchange", onFsChange);
   325→   325→    return () => document.removeEventListener("fullscreenchange", onFsChange);
   326→   326→  }, []);
   327→   327→
   328→   328→  // Keyboard navigation & remote-control shortcuts.
   329→   329→  useEffect(() => {
   330→   330→    const handleKeyDown = (e: KeyboardEvent) => {
   331→   331→      if (["INPUT", "TEXTAREA"].includes((e.target as HTMLElement)?.tagName)) return;
   332→   332→
   333→   333→      if (e.key === "Escape") {
   334→   334→        // If we're in fullscreen, let the browser exit fullscreen first;
   335→   335→        // only close the modal when not in fullscreen.
   336→   336→        if (!document.fullscreenElement) {
   337→   337→          e.preventDefault();
   338→   338→          onClose();
   339→   339→        }
   340→   340→      } else if (e.key === " " || e.key === "Enter") {
   341→   341→        e.preventDefault();
   342→   342→        togglePlay();
   343→   343→      } else if (e.key === "f" || e.key === "F") {
   344→   344→        e.preventDefault();
   345→   345→        toggleFullscreen();
   346→   346→      } else if (e.key === "ArrowUp") {
   347→   347→        e.preventDefault();
   348→   348→        const newVol = Math.min(100, volume + 10);
   349→   349→        setVolume(newVol);
   350→   350→        setIsMuted(false);
   351→   351→        if (videoRef.current) {
   352→   352→          videoRef.current.volume = newVol / 100;
   353→   353→          videoRef.current.muted = false;
   354→   354→        }
   355→   355→      } else if (e.key === "ArrowDown") {
   356→   356→        e.preventDefault();
   357→   357→        const newVol = Math.max(0, volume - 10);
   358→   358→        setVolume(newVol);
   359→   359→        if (videoRef.current) {
   360→   360→          videoRef.current.volume = newVol / 100;
   361→   361→        }
   362→   362→      } else if (e.key === "m" || e.key === "M") {
   363→   363→        e.preventDefault();
   364→   364→        toggleMute();
   365→   365→      } else if (e.key === "ArrowLeft") {
   366→   366→        e.preventDefault();
   367→   367→        skipBy(-10);
   368→   368→      } else if (e.key === "ArrowRight") {
   369→   369→        e.preventDefault();
   370→   370→        skipBy(10);
   371→   371→      }
   372→   372→    };
   373→   373→
   374→   374→    window.addEventListener("keydown", handleKeyDown);
   375→   375→    return () => window.removeEventListener("keydown", handleKeyDown);
   376→   376→  }, [volume, togglePlay, toggleFullscreen, toggleMute, skipBy, onClose]);
   377→   377→
   378→   378→  // Server label: proxy fallbacks get the "Lecteur Sécurisé (proxy)" label,
   379→   379→  // direct-video servers show their hoster name.
   380→   380→  const serverLabel = (srv: StreamServer): string => {
   381→   381→    if (srv.videoUrl && !isDirectVideo(srv.videoUrl)) {
   382→   382→      return "Lecteur Sécurisé (proxy)";
   383→   383→    }
   384→   384→    return srv.name || srv.hoster;
   385→   385→  };
   386→   386→
   387→   387→  // Timeline fill percentage (guard against div-by-zero before metadata loads).
   388→   388→  const timelinePct = duration > 0 ? (currentTime / duration) * 100 : 0;
   389→   389→
   390→   390→  return (
   391→   391→    <div
   392→   392→      className="fixed inset-0 z-50 flex items-center justify-center p-2 sm:p-4 bg-black/90 backdrop-blur-md animate-fade-in"
   393→   393→      onClick={onClose}
   394→   394→    >
   395→   395→      <div
   396→   396→        ref={containerRef}
   397→   397→        className="w-full max-w-6xl bg-surface-container-high rounded-2xl overflow-hidden shadow-2xl border border-outline-variant/20 flex flex-col h-[94vh]"
   398→   398→        onClick={(e) => e.stopPropagation()}
   399→   399→      >
   400→   400→        {/* Secure In-App Browser Top Action Bar (URL bar removed) */}
   401→   401→        <div className="bg-surface-container-highest px-3 py-2 border-b border-outline-variant/15 flex items-center justify-between gap-2 flex-wrap">
   402→   402→          {/* Left: Navigation Controls (back & refresh) + uBlock blocked-ads pill */}
   403→   403→          <div className="flex items-center gap-1 text-on-surface-variant shrink-0">
   404→   404→            <button
   405→   405→              onClick={onClose}
   406→   406→              className="w-8 h-8 rounded-full hover:bg-surface-container flex items-center justify-center transition-colors cursor-pointer"
   407→   407→              title="Fermer (Échap)"
   408→   408→            >
   409→   409→              <span className="material-symbols-outlined text-[18px]">arrow_back</span>
   410→   410→            </button>
   411→   411→            <button
   412→   412→              onClick={handleRefresh}
   413→   413→              className="w-8 h-8 rounded-full hover:bg-surface-container flex items-center justify-center transition-colors cursor-pointer"
   414→   414→              title="Actualiser et re-filtrer les pubs"
   415→   415→            >
   416→   416→              <span className="material-symbols-outlined text-[18px]">refresh</span>
   417→   417→            </button>
   418→   418→            <div className="flex items-center gap-1.5 ml-2 px-2.5 py-1 rounded-full bg-primary/10 border border-primary/30 text-primary text-[11px] font-semibold">
   419→   419→              <span className="material-symbols-outlined text-[14px]">shield</span>
   420→   420→              <span className="font-bold">{blockedAdsOnPage} pubs bloquées</span>
   421→   421→              <span className="w-1.5 h-1.5 rounded-full bg-primary animate-pulse" />
   422→   422→            </div>
   423→   423→          </div>
   424→   424→
   425→   425→          {/* Right: Action Buttons (uBlock, download, favorite, fullscreen, close) */}
   426→   426→          <div className="flex items-center gap-2">
   427→   427→            {/* uBlock Origin Shield Trigger */}
   428→   428→            <button
   429→   429→              onClick={() => setShowUBlockModal(true)}
   430→   430→              className="flex items-center gap-1.5 bg-primary/10 hover:bg-primary/20 text-primary text-[11px] font-semibold px-2.5 py-1.5 rounded-full border border-primary/30 transition-all cursor-pointer"
   431→   431→              title="Ouvrir le panneau uBlock Origin"
   432→   432→            >
   433→   433→              <span className="material-symbols-outlined text-[15px]">shield</span>
   434→   434→              <span className="font-bold hidden md:inline">uBlock</span>
   435→   435→              <span className="w-1.5 h-1.5 rounded-full bg-primary animate-pulse" />
   436→   436→            </button>
   437→   437→
