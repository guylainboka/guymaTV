    67→/**
    68→ * Returns true when `url` is a direct video file we can hand to a native
    69→ * `<video>` element (.mp4 / .m3u8 / .webm, or any absolute http(s) URL that
    70→ * isn't our own /api/proxy fallback). Returns false for the proxy URL.
    71→ */
    72→function isDirectVideo(url: string): boolean {
    73→  if (!url) return false;
    74→  if (url.startsWith("/api/proxy")) return false;
    75→  if (/^https?:\/\//i.test(url)) return true;
    76→  return /\.(mp4|m3u8|webm)(\?|$)/i.test(url);
    77→}

   106→interface CommentItemProps {
   107→  comment: CommentDTO;
   108→  isReply?: boolean;
   109→  replyingTo: string | null;

   115→const CommentItem: React.FC<CommentItemProps> = ({
   116→  comment,
   117→  isReply = false,
   118→  replyingTo,
   119→  onToggleReply,
   120→  onSubmitReply,
   121→  onDelete,
   122→}) => {
   123→  const [replyText, setReplyText] = useState("");
   124→  const initials = (comment.userName || "A").charAt(0).toUpperCase();
   125→  const formattedDate = new Date(comment.createdAt).toLocaleDateString("fr-FR", {
   126→    day: "numeric",
   127→    month: "short",
   128→    year: "numeric",
   129→  });
   130→  const showReplyForm = replyingTo === comment.id;
   131→
   132→  const handleSubmit = () => {
   133→    if (!replyText.trim()) return;
   134→    onSubmitReply(comment.id, replyText);
   135→    setReplyText("");
   136→  };
   137→
   138→  return (
   139→    <div className={isReply ? "ml-10 sm:ml-12" : ""}>
   140→      <div className="bg-surface-container-low rounded-lg p-3 border border-outline-variant/10">
   141→        <div className="flex items-start gap-3">
   142→          <div className="w-8 h-8 rounded-full bg-primary/20 text-primary flex items-center justify-center font-bold text-body-sm shrink-0">
   143→            {initials}
   144→          </div>
   145→          <div className="flex-1 min-w-0">
   146→            <div className="flex items-center justify-between gap-2 flex-wrap">
   147→              <div className="flex items-center gap-2">
   148→                <span className="font-bold text-on-surface text-body-sm">
   149→                  {comment.userName}
   150→                </span>
   151→                {comment.rating > 0 && <StarRow rating={comment.rating} size={11} />}
   152→              </div>
   153→              <span className="text-label-sm text-outline">{formattedDate}</span>
   154→            </div>
   155→            <p className="text-body-sm text-on-surface-variant mt-1 whitespace-pre-wrap break-words">
   156→              {comment.content}
   157→            </p>
   158→            <div className="flex items-center gap-3 mt-2">
   159→              {!isReply && (
   160→                <button
   161→                  onClick={() => onToggleReply(comment.id)}
   162→                  className="text-label-sm text-primary font-semibold flex items-center gap-1 hover:underline cursor-pointer"
   163→                >
   164→                  <span className="material-symbols-outlined text-[14px]">reply</span>
   165→                  Répondre
   166→                </button>
   167→              )}
   168→              <button
   169→                onClick={() => onDelete(comment.id)}
   170→                className="text-label-sm text-outline hover:text-error flex items-center gap-1 cursor-pointer ml-auto"
   171→                title="Supprimer ce commentaire"
   172→              >
   173→                <span className="material-symbols-outlined text-[14px]">delete</span>
   174→              </button>
   175→            </div>
   176→          </div>
   177→        </div>
   178→      </div>
   179→
   180→      {/* Inline reply form */}
   181→      {showReplyForm && (
   182→        <div className="mt-2 ml-11 bg-surface-container-low rounded-lg p-2 border border-outline-variant/10 flex gap-2 items-end">
   183→          <textarea
   184→            value={replyText}
   185→            onChange={(e) => setReplyText(e.target.value)}
   186→            placeholder="Votre réponse..."
   187→            rows={2}
   188→            className="flex-1 p-2 bg-surface-container text-on-surface text-body-sm rounded outline-none focus:ring-2 focus:ring-primary border border-outline-variant/20 resize-none"
   189→          />
   190→          <button
   191→            onClick={handleSubmit}
   192→            disabled={!replyText.trim()}
   193→            className="px-3 py-1.5 rounded-lg bg-primary text-on-primary text-label-sm font-bold flex items-center gap-1 hover:scale-105 transition-transform cursor-pointer disabled:opacity-50 disabled:cursor-not-allowed"
   194→            title="Publier la réponse"
   195→          >
   196→            <span className="material-symbols-outlined text-[14px]">send</span>
   197→          </button>
   198→        </div>
   199→      )}
   200→
   201→      {/* Nested replies */}
   202→      {comment.replies && comment.replies.length > 0 && (
   203→        <div className="mt-2 space-y-2">
   204→          {comment.replies.map((reply) => (
   205→            <CommentItem
   206→              key={reply.id}
   207→              comment={reply}
   208→              isReply
   209→              replyingTo={replyingTo}
   210→              onToggleReply={onToggleReply}
   211→              onSubmitReply={onSubmitReply}
   212→              onDelete={onDelete}
   213→            />
   214→          ))}
   215→        </div>
   216→      )}
   217→    </div>
   218→  );
   219→};

   221→/**
   222→ * Secure in-app stream player modal — full reading page with hero, video
   223→ * player, info section, series episodes section (if series), and comments &
   224→ * reviews section.
   225→ *
   226→ * Ported from the original maquette with the following adjustments:
   227→ *  - The fake URL/address bar (lock icon, domain, path, "SSL Sécurisé" badge)
   228→ *    is REMOVED entirely so only the video and player controls remain.
   229→ *  - On mount we fetch the real extracted servers from
   230→ *    `GET /api/extract?id=${item.id}` (Playwright, cached 30 min server-side).
   231→ *    A spinner with "Extraction des lecteurs en cours…" is shown while
   232→ *    Playwright runs (~10-15s on first call).
   233→ *  - Smart rendering: if `currentServer.videoUrl` is a direct video URL
   234→ *    (mp4/m3u8/webm or any http(s) URL that isn't /api/proxy), we render a
   235→ *    native `<video>` element and wire every player control to it via refs
   236→ *    (play/pause, timeline scrub, volume, skip ±10s, fullscreen). If the URL
   237→ *    is `/api/proxy?page=...` (extraction failed/empty), we fall back to the
   238→ *    filtered `<iframe>` as before.
   239→ *  - The "Site Complet (Nettoyé)" view mode toggle is removed — only the
   240→ *    cinematic player view remains.
   241→ *  - NEW: Series episodes section — if `item.category === "Séries"`, fetches
   242→ *    `/api/series-structure?id=${item.id}` on mount and renders a season
   243→ *    selector + episode list (VF episodes prioritized). Selecting an episode
   244→ *    sets it as the current video source.
   245→ *  - NEW: Comments & reviews section — always shows. Fetches
   246→ *    `/api/comments?streamItemId=${item.id}`, supports star ratings, replies,
   247→ *    and deletion. After POST, refreshes the list.
   248→ *  - The maquette chrome (top action bar, uBlock badge, player stage, bottom
   249→ *    control bar, server selector, media details, uBlock protection banner)
   250→ *    is preserved identically.
   251→ */
   252→export const SecureBrowserModal: React.FC<SecureBrowserModalProps> = ({
   253→  item,
   254→  onClose,
   255→  onStartDownload,
   256→  isFavorite,
   257→  onToggleFavorite,
   258→  onSelectRelatedItem,
   259→}) => {
   260→  // ---- Server fetch state (populated from /api/extract on mount) ----
   261→  const [servers, setServers] = useState<StreamServer[]>([]);
   262→  const [serversLoading, setServersLoading] = useState(true);
   263→
   264→  // ---- Player state ----
   265→  const [selectedServerIndex, setSelectedServerIndex] = useState(0);
   266→  const [selectedEpisode, setSelectedEpisode] = useState<SeriesEpisode | null>(null);
   267→  const [isPlaying, setIsPlaying] = useState(true);
   268→  const [isFullscreen, setIsFullscreen] = useState(false);
   269→  const [currentTime, setCurrentTime] = useState(0);
   270→  const [duration, setDuration] = useState(0);
   271→  const [volume, setVolume] = useState(85);
   272→  const [isMuted, setIsMuted] = useState(false);
   273→  const [showUBlockModal, setShowUBlockModal] = useState(false);
   274→  const [blockedAdsOnPage, setBlockedAdsOnPage] = useState(
   275→    (item.blockedAdStats?.popups || 12) + (item.blockedAdStats?.adultBanners || 6)
   276→  );
   277→  const [iframeKey, setIframeKey] = useState(0);
   278→  const [switchingServer, setSwitchingServer] = useState(false);
   279→
   280→  // ---- Series structure state (fetched from /api/series-structure) ----
   281→  const [seriesStructure, setSeriesStructure] = useState<SeriesStructure | null>(null);
   282→  const [seriesLoading, setSeriesLoading] = useState(false);
   283→  const [selectedSeason, setSelectedSeason] = useState(1);
   284→
   285→  // ---- Comments state (fetched from /api/comments) ----
   286→  const [comments, setComments] = useState<CommentDTO[]>([]);
   287→  const [commentsLoading, setCommentsLoading] = useState(true);
   288→  const [averageRating, setAverageRating] = useState(0);
   289→  const [newCommentName, setNewCommentName] = useState("");
   290→  const [newCommentRating, setNewCommentRating] = useState(0);
   291→  const [hoverRating, setHoverRating] = useState(0);
   292→  const [newCommentContent, setNewCommentContent] = useState("");
   293→  const [submittingComment, setSubmittingComment] = useState(false);
   294→  const [replyingTo, setReplyingTo] = useState<string | null>(null);
   295→
   296→  const containerRef = useRef<HTMLDivElement>(null);
   297→  const videoRef = useRef<HTMLVideoElement>(null);
   298→
   299→  // Detect series: item.category === "Séries" OR item.episodes already populated.
   300→  const isSeries =
   301→    item.category === "Séries" ||
   302→    item.category === "Animés" ||
   303→    (Array.isArray(item.episodes) && item.episodes.length > 0);
   304→
   305→  // Build the proxy fallback server (used when extraction fails or returns []).
   306→  const buildProxyFallback = useCallback((): StreamServer => ({
   307→    id: `srv-${item.id}-secure`,
   308→    name: "Lecteur Sécurisé (proxy)",
   309→    hoster: "Direct 4K",
   310→    quality: item.quality,
   311→    language: "VF",
   312→    speed: "Ultra Rapide (sans pub)",
   313→    videoUrl: apiUrl(`/api/proxy?page=${encodeURIComponent(item.id)}`),
   314→  }), [item.id, item.quality]);
   315→
   316→  // Fetch real extracted servers from /api/extract on mount (and when item.id changes).
   317→  useEffect(() => {
   318→    let cancelled = false;
   319→    setServersLoading(true);
   320→    setSwitchingServer(false);
   321→    setSelectedServerIndex(0);
   322→    setSelectedEpisode(null);
   323→    setCurrentTime(0);
   324→    setDuration(0);
   325→    setIsPlaying(true);
   326→
   327→    fetch(apiUrl(`/api/extract?id=${encodeURIComponent(item.id)}`))
   328→      .then((r) => {
   329→        if (!r.ok) throw new Error(`extract failed: ${r.status}`);
   330→        return r.json();
   331→      })
   332→      .then((data: StreamServer[] | { servers?: StreamServer[] }) => {
   333→        if (cancelled) return;
   334→        const list: StreamServer[] = Array.isArray(data)
   335→          ? data
   336→          : (data?.servers || []);
   337→        if (list.length > 0) {
   338→          setServers(list);
   339→        } else {
   340→          // Empty result — fall back to a single proxy server.
   341→          setServers([buildProxyFallback()]);
   342→        }
   343→      })
   344→      .catch(() => {
   345→        if (cancelled) return;
   346→        // Network/extractor error — fall back to a single proxy server.
   347→        setServers([buildProxyFallback()]);
   348→      })
   349→      .finally(() => {
   350→        if (!cancelled) setServersLoading(false);
   351→      });
   352→
   353→    return () => { cancelled = true; };
   354→  }, [item.id, buildProxyFallback]);
   355→
   356→  // Fetch series structure (only if item is a series).
   357→  useEffect(() => {
   358→    if (!isSeries) {
   359→      setSeriesStructure(null);
   360→      return;
   361→    }
   362→    let cancelled = false;
   363→    setSeriesLoading(true);
   364→
   365→    fetch(apiUrl(`/api/series-structure?id=${encodeURIComponent(item.id)}`))
   366→      .then((r) => {
   367→        if (!r.ok) throw new Error(`series-structure failed: ${r.status}`);
   368→        return r.json();
   369→      })
   370→      .then((data: SeriesStructure) => {
   371→        if (cancelled) return;
   372→        setSeriesStructure(data);
   373→        // Default to the current season (or 1 if none)
   374→        setSelectedSeason(data.currentSeason || data.seasons?.[0]?.seasonNumber || 1);
   375→      })
   376→      .catch(() => {
   377→        if (cancelled) return;
   378→        setSeriesStructure(null);
   379→      })
   380→      .finally(() => {
   381→        if (!cancelled) setSeriesLoading(false);
   382→      });
   383→
   384→    return () => { cancelled = true; };
   385→    // eslint-disable-next-line react-hooks/exhaustive-deps
   386→  }, [item.id, isSeries]);
   387→
   388→  // Fetch comments on mount (always).
   389→  // `refreshComments` is also called after every POST/DELETE to keep the list
   390→  // in sync. The cancellation flag only applies to the most recent call, so
   391→  // late state updates from a stale fetch are silently ignored.
   392→  const commentsCancelRef = useRef<() => void>(() => {});
   393→  const refreshComments = useCallback(() => {
   394→    // Cancel any in-flight comment fetch first.
   395→    commentsCancelRef.current();
   396→    let cancelled = false;
   397→    commentsCancelRef.current = () => { cancelled = true; };
   398→    setCommentsLoading(true);
   399→    fetch(apiUrl(`/api/comments?streamItemId=${encodeURIComponent(item.id)}`))
   400→      .then((r) => {
   401→        if (!r.ok) throw new Error(`comments failed: ${r.status}`);
   402→        return r.json();
   403→      })
   404→      .then((data: { comments: CommentDTO[]; total: number; averageRating: number }) => {
   405→        if (cancelled) return;
   406→        setComments(data.comments || []);
   407→        setAverageRating(data.averageRating || 0);
   408→      })
   409→      .catch(() => {
   410→        if (cancelled) return;
   411→        setComments([]);
   412→        setAverageRating(0);
   413→      })
   414→      .finally(() => {
   415→        if (!cancelled) setCommentsLoading(false);
   416→      });
   417→  }, [item.id]);
   418→
   419→  useEffect(() => {
   420→    refreshComments();
   421→    // On unmount, cancel the in-flight comment fetch.
   422→    return () => commentsCancelRef.current();
   423→  }, [refreshComments]);
   424→
   425→  const currentServer = servers[selectedServerIndex] || servers[0];
   426→
   427→  // Resolve the URL to actually play. Selected episode wins (for series),
   428→  // otherwise the current server's videoUrl, otherwise the proxy.
   429→  const currentVideoUrl: string =
   430→    selectedEpisode?.videoUrl ||
   431→    currentServer?.videoUrl ||
   432→    apiUrl(`/api/proxy?page=${encodeURIComponent(item.id)}`);
   433→
   434→  const directVideo = isDirectVideo(currentVideoUrl);
   435→  const isHlsStream = directVideo && /\.m3u8(\?|$)/i.test(currentVideoUrl);
   436→
   437→  // Attach HLS.js to the <video> element when the stream is a .m3u8 URL.
   438→  // Native HLS (Safari/iOS) doesn't need hls.js — it plays .m3u8 directly.
   439→  useEffect(() => {
   440→    if (!directVideo || !isHlsStream) return;
   441→    const video = videoRef.current;
   442→    if (!video) return;
   443→
   444→    let hls: import("hls.js").default | null = null;
   445→
   446→    // Native HLS support (Safari, iOS)
   447→    if (video.canPlayType("application/vnd.apple.mpegurl")) {
   448→      video.src = currentVideoUrl;
   449→      video.play().catch(() => {});
   450→      return () => {
   451→        video.removeAttribute("src");
   452→        video.load();
   453→      };
   454→    }
   455→
   456→    // Other browsers → use hls.js
   457→    let cancelled = false;
   458→    (async () => {
   459→      if (!HlsModule) {
   460→        HlsModule = (await import("hls.js")).default;
   461→      }
   462→      if (cancelled || !HlsModule) return;
   463→      if (!HlsModule.isSupported()) return;
   464→      hls = new HlsModule({ enableWorker: true, lowLatencyMode: false });
   465→      hls.loadSource(currentVideoUrl);
   466→      hls.attachMedia(video);
   467→      hls.on(HlsModule.Events.MANIFEST_PARSED, () => {
   468→        video.play().catch(() => {});
   469→        setSwitchingServer(false);
   470→      });
   471→      hls.on(HlsModule.Events.ERROR, (_evt, data) => {
   472→        if (data.fatal) {
   473→          console.error("[hls.js] fatal error:", data);
   474→          setSwitchingServer(false);
   475→        }
   476→      });
   477→    })();
   478→
   479→    return () => {
   480→      cancelled = true;
   481→      if (hls) {
   482→        hls.destroy();
   483→      }
   484→    };
   485→  }, [currentVideoUrl, directVideo, isHlsStream]);
   486→
   487→  const domain = item.sourceServiceId
   488→    ? `${item.sourceServiceId}.net`
   489→    : "french-stream.net";
   490→
   491→  // =========================================================================
   492→  // Native <video> controls — every control below is wired to videoRef.
   493→  // =========================================================================
   494→
   495→  const togglePlay = useCallback(() => {
   496→    const v = videoRef.current;
   497→    if (!v) {
   498→      setIsPlaying((prev) => !prev);
   499→      return;
   500→    }
   501→    if (v.paused) {
   502→      v.play().then(() => setIsPlaying(true)).catch(() => {});
   503→    } else {
   504→      v.pause();
   505→      setIsPlaying(false);
   506→    }
   507→  }, []);
   508→
   509→  const toggleFullscreen = useCallback(() => {
   510→    if (!containerRef.current) return;
   511→    if (!document.fullscreenElement) {
   512→      containerRef.current
   513→        .requestFullscreen?.()
   514→        .then(() => setIsFullscreen(true))
   515→        .catch(() => {});
   516→    } else {
   517→      document.exitFullscreen?.()
   518→        .then(() => setIsFullscreen(false))
   519→        .catch(() => {});
   520→    }
   521→  }, []);
   522→
   523→  const skipBy = useCallback((delta: number) => {
   524→    const v = videoRef.current;
   525→    if (!v) return;
   526→    const target = Math.max(0, Math.min(v.duration || 0, v.currentTime + delta));
   527→    v.currentTime = target;
   528→    setCurrentTime(target);
   529→  }, []);
   530→
   531→  const handleSeek = (e: React.MouseEvent<HTMLDivElement>) => {
   532→    const v = videoRef.current;
   533→    if (!v || !v.duration) return;
   534→    const rect = e.currentTarget.getBoundingClientRect();
   535→    const pos = (e.clientX - rect.left) / rect.width;
   536→    const newTime = Math.max(0, Math.min(v.duration, pos * v.duration));
   537→    v.currentTime = newTime;
   538→    setCurrentTime(newTime);
   539→  };
   540→
   541→  const handleVolumeChange = (e: React.ChangeEvent<HTMLInputElement>) => {
   542→    const newVol = Number(e.target.value);
   543→    setVolume(newVol);
   544→    setIsMuted(false);
   545→    const v = videoRef.current;
   546→    if (v) {
   547→      v.volume = newVol / 100;
   548→      v.muted = false;
   549→    }
   550→  };
   551→
   552→  const toggleMute = useCallback(() => {
   553→    const v = videoRef.current;
   554→    if (!v) {
   555→      setIsMuted((p) => !p);
