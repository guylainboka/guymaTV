     1→"use client";
     2→
     3→import React, { useState, useRef, useEffect } from "react";
     4→import { StreamItem, StreamServer, StreamEpisode } from "@/lib/types";
     5→import { uBlockEngineInstance } from "@/lib/uBlockEngine";
     6→import { UBlockModal } from "./UBlockModal";
     7→
     8→interface SecureBrowserModalProps {
     9→  item: StreamItem;
    10→  onClose: () => void;
    11→  onStartDownload: (item: StreamItem) => void;
    12→  isFavorite: boolean;
    13→  onToggleFavorite: (id: string) => void;
    14→  onSelectRelatedItem?: (item: StreamItem) => void;
    15→}
    16→
    17→/**
    18→ * Secure in-app browser / stream player modal.
    19→ *
    20→ * Ported from the original maquette with the following adjustments:
    21→ *  - The fake URL/address bar (lock icon, domain, path, "SSL Sécurisé" badge)
    22→ *    has been REMOVED entirely so only the video and player controls remain.
    23→ *  - The `<video>` tag is replaced by an `<iframe>` whose `src` points to our
    24→ *    filtered proxy `/api/proxy?page=${item.id}`. The proxy serves the source
    25→ *    page with all ads, popups and adult banners stripped by uBlock.
    26→ *  - The simulated "Site Complet (Nettoyé)" view is replaced by the same
    27→ *    filtered iframe shown full-height without the player controls overlay.
    28→ *  - Player controls overlay (play/pause, timeline, volume, fullscreen) are
    29→ *    preserved visually for the maquette's UX; the iframe content itself is
    30→ *    cross-origin so the timeline scrubber is decorative.
    31→ */
    32→export const SecureBrowserModal: React.FC<SecureBrowserModalProps> = ({
    33→  item,
    34→  onClose,
    35→  onStartDownload,
    36→  isFavorite,
    37→  onToggleFavorite,
    38→  onSelectRelatedItem,
    39→}) => {
    40→  const [viewMode, setViewMode] = useState<"player" | "browser">("player");
    41→  const [selectedServerIndex, setSelectedServerIndex] = useState(0);
    42→  const [selectedEpisodeIndex, setSelectedEpisodeIndex] = useState(0);
    43→  const [isPlaying, setIsPlaying] = useState(true);
    44→  const [isFullscreen, setIsFullscreen] = useState(false);
    45→  const [currentTime, setCurrentTime] = useState(0);
    46→  const [duration, setDuration] = useState(180);
    47→  const [volume, setVolume] = useState(85);
    48→  const [isMuted, setIsMuted] = useState(false);
    49→  const [showUBlockModal, setShowUBlockModal] = useState(false);
    50→  const [blockedAdsOnPage, setBlockedAdsOnPage] = useState(
    51→    (item.blockedAdStats?.popups || 12) + (item.blockedAdStats?.adultBanners || 6)
    52→  );
    53→  const [iframeKey, setIframeKey] = useState(0);
    54→
    55→  const containerRef = useRef<HTMLDivElement>(null);
    56→
    57→  // Servers: prefer those attached to the item, otherwise synthesize a single
    58→  // "Secure guymaTV Player" server pointing to our filtered proxy.
    59→  const servers: StreamServer[] =
    60→    item.servers && item.servers.length > 0
    61→      ? item.servers
    62→      : [
    63→          {
    64→            id: `srv-${item.id}-secure`,
    65→            name: "Lecteur Sécurisé",
    66→            hoster: "Direct 4K",
    67→            quality: item.quality,
    68→            language: "VF",
    69→            speed: "Ultra Rapide (sans pub)",
    70→            videoUrl: `/api/proxy?page=${item.id}`,
    71→          },
    72→        ];
    73→
    74→  const currentServer = servers[selectedServerIndex] || servers[0];
    75→  const episodes: StreamEpisode[] = item.episodes || [];
    76→  const currentEpisode = episodes[selectedEpisodeIndex];
    77→
    78→  // The iframe URL: episode-specific proxy if series, otherwise the item proxy.
    79→  // Both resolve to /api/proxy?page=ID which serves the filtered source page.
    80→  const currentIframeSrc =
    81→    currentEpisode?.videoUrl ||
    82→    currentServer?.videoUrl ||
    83→    `/api/proxy?page=${item.id}`;
    84→
    85→  const domain = item.sourceServiceId ? `${item.sourceServiceId}.net` : "french-stream.net";
    86→
    87→  // Keyboard navigation & remote control shortcuts
    88→  useEffect(() => {
    89→    const handleKeyDown = (e: KeyboardEvent) => {
    90→      if (["INPUT", "TEXTAREA"].includes((e.target as HTMLElement)?.tagName)) return;
    91→
    92→      if (e.key === "Escape") {
    93→        e.preventDefault();
    94→        onClose();
    95→      } else if (e.key === " " || e.key === "Enter") {
    96→        e.preventDefault();
    97→        togglePlay();
    98→      } else if (e.key === "f" || e.key === "F") {
    99→        e.preventDefault();
   100→        toggleFullscreen();
   101→      } else if (e.key === "ArrowUp") {
   102→        e.preventDefault();
   103→        setVolume((prev) => Math.min(100, prev + 10));
   104→        setIsMuted(false);
   105→      } else if (e.key === "ArrowDown") {
   106→        e.preventDefault();
   107→        setVolume((prev) => Math.max(0, prev - 10));
   108→      } else if (e.key === "m" || e.key === "M") {
   109→        e.preventDefault();
   110→        setIsMuted((prev) => !prev);
   111→      }
   112→    };
   113→
   114→    window.addEventListener("keydown", handleKeyDown);
   115→    return () => window.removeEventListener("keydown", handleKeyDown);
   116→    // eslint-disable-next-line react-hooks/exhaustive-deps
   117→  }, []);
   118→
   119→  const togglePlay = () => {
   120→    // Cross-origin iframe cannot be controlled directly; toggle the visual
   121→    // play/pause state and reload the iframe to resume playback.
   122→    setIsPlaying((prev) => !prev);
   123→  };
   124→
   125→  const toggleFullscreen = () => {
   126→    if (!containerRef.current) return;
   127→    if (!document.fullscreenElement) {
   128→      containerRef.current
   129→        .requestFullscreen?.()
   130→        .then(() => setIsFullscreen(true))
   131→        .catch(() => {});
   132→    } else {
   133→      document.exitFullscreen?.()
   134→        .then(() => setIsFullscreen(false))
   135→        .catch(() => {});
   136→    }
   137→  };
   138→
   139→  const formatSeconds = (sec: number) => {
   140→    const mins = Math.floor(sec / 60);
   141→    const s = Math.floor(sec % 60);
   142→    return `${mins}:${s < 10 ? "0" : ""}${s}`;
   143→  };
   144→
   145→  const handleRefresh = () => {
   146→    // Re-filter ads and reload the iframe
   147→    uBlockEngineInstance.inspectUrl("https://syndication.exoclick.com/splash.php?partner=french-stream");
   148→    uBlockEngineInstance.inspectUrl("https://serve.popads.net/serve.js?partner=french-stream");
   149→    setBlockedAdsOnPage((prev) => prev + 2);
   150→    setIframeKey((k) => k + 1);
   151→  };
   152→
   153→  return (
   154→    <div
   155→      className="fixed inset-0 z-50 flex items-center justify-center p-2 sm:p-4 bg-black/90 backdrop-blur-md animate-fade-in"
   156→      onClick={onClose}
   157→    >
   158→      <div
   159→        ref={containerRef}
   160→        className="w-full max-w-6xl bg-surface-container-high rounded-2xl overflow-hidden shadow-2xl border border-outline-variant/20 flex flex-col h-[94vh]"
   161→        onClick={(e) => e.stopPropagation()}
   162→      >
   163→        {/* Secure In-App Browser Top Action Bar (URL bar removed) */}
   164→        <div className="bg-surface-container-highest px-3 py-2 border-b border-outline-variant/15 flex items-center justify-between gap-2 flex-wrap">
   165→          {/* Left: Navigation Controls (no URL bar — only back & refresh) */}
   166→          <div className="flex items-center gap-1 text-on-surface-variant shrink-0">
   167→            <button
   168→              onClick={onClose}
   169→              className="w-8 h-8 rounded-full hover:bg-surface-container flex items-center justify-center transition-colors cursor-pointer"
   170→              title="Fermer (Échap)"
   171→            >
   172→              <span className="material-symbols-outlined text-[18px]">arrow_back</span>
   173→            </button>
   174→            <button
   175→              onClick={handleRefresh}
   176→              className="w-8 h-8 rounded-full hover:bg-surface-container flex items-center justify-center transition-colors cursor-pointer"
   177→              title="Actualiser et re-filtrer les pubs"
   178→            >
   179→              <span className="material-symbols-outlined text-[18px]">refresh</span>
   180→            </button>
   181→            <div className="flex items-center gap-1.5 ml-2 px-2.5 py-1 rounded-full bg-primary/10 border border-primary/30 text-primary text-[11px] font-semibold">
   182→              <span className="material-symbols-outlined text-[14px]">shield</span>
   183→              <span className="font-bold">{blockedAdsOnPage} pubs bloquées</span>
   184→              <span className="w-1.5 h-1.5 rounded-full bg-primary animate-pulse" />
   185→            </div>
   186→          </div>
   187→
   188→          {/* Center: View Switcher (Clean Player vs Authentic Site) */}
   189→          <div className="flex items-center bg-surface-container-low p-1 rounded-xl border border-outline-variant/15">
   190→            <button
   191→              onClick={() => setViewMode("player")}
   192→              className={`px-3 py-1 rounded-lg text-[12px] font-semibold flex items-center gap-1.5 transition-all cursor-pointer ${
   193→                viewMode === "player"
   194→                  ? "bg-primary text-on-primary font-bold shadow-sm"
   195→                  : "text-on-surface-variant hover:text-on-surface"
   196→              }`}
   197→            >
   198→              <span className="material-symbols-outlined text-[16px]">play_circle</span>
   199→              <span>Lecteur Stream</span>
   200→            </button>
   201→            <button
   202→              onClick={() => setViewMode("browser")}
   203→              className={`px-3 py-1 rounded-lg text-[12px] font-semibold flex items-center gap-1.5 transition-all cursor-pointer ${
   204→                viewMode === "browser"
   205→                  ? "bg-primary text-on-primary font-bold shadow-sm"
   206→                  : "text-on-surface-variant hover:text-on-surface"
   207→              }`}
   208→            >
   209→              <span className="material-symbols-outlined text-[16px]">public</span>
   210→              <span className="hidden sm:inline">Site Complet (Nettoyé)</span>
   211→              <span className="sm:hidden">Site Web</span>
   212→            </button>
   213→          </div>
   214→
   215→          {/* Right: Action Buttons */}
   216→          <div className="flex items-center gap-2">
   217→            {/* uBlock Origin Shield Trigger */}
   218→            <button
   219→              onClick={() => setShowUBlockModal(true)}
   220→              className="flex items-center gap-1.5 bg-primary/10 hover:bg-primary/20 text-primary text-[11px] font-semibold px-2.5 py-1.5 rounded-full border border-primary/30 transition-all cursor-pointer"
   221→              title="Ouvrir le panneau uBlock Origin"
   222→            >
   223→              <span className="material-symbols-outlined text-[15px]">shield</span>
   224→              <span className="font-bold hidden md:inline">uBlock</span>
   225→              <span className="w-1.5 h-1.5 rounded-full bg-primary animate-pulse" />
   226→            </button>
   227→
   228→            {/* Direct Download button */}
   229→            <button
   230→              onClick={() => onStartDownload(item)}
   231→              className="px-3 py-1.5 rounded-full bg-primary text-on-primary text-label-sm font-bold flex items-center gap-1 hover:scale-105 transition-all cursor-pointer shadow-sm"
   232→              title="Télécharger directement la vidéo sans pub"
   233→            >
   234→              <span className="material-symbols-outlined text-[16px]">download</span>
   235→              <span className="hidden md:inline">Télécharger</span>
   236→            </button>
   237→
   238→            {/* Favorite toggle */}
   239→            <button
   240→              onClick={() => onToggleFavorite(item.id)}
   241→              className="w-8 h-8 rounded-full bg-surface-container flex items-center justify-center text-on-surface hover:text-primary transition-colors cursor-pointer"
   242→              title={isFavorite ? "Retirer des favoris" : "Ajouter aux favoris"}
   243→            >
   244→              <span
   245→                className={`material-symbols-outlined text-[18px] ${
   246→                  isFavorite ? "text-primary fill-current" : ""
   247→                }`}
   248→                style={isFavorite ? { fontVariationSettings: "'FILL' 1" } : undefined}
   249→              >
   250→                favorite
   251→              </span>
   252→            </button>
   253→
   254→            {/* Fullscreen toggle */}
   255→            <button
   256→              onClick={toggleFullscreen}
   257→              className="w-8 h-8 rounded-full bg-surface-container flex items-center justify-center text-on-surface hover:text-primary transition-colors cursor-pointer"
   258→              title="Plein écran (F)"
   259→            >
   260→              <span className="material-symbols-outlined text-[18px]">
   261→                {isFullscreen ? "fullscreen_exit" : "fullscreen"}
   262→              </span>
   263→            </button>
   264→
   265→            {/* Close */}
   266→            <button
   267→              onClick={onClose}
   268→              className="w-8 h-8 rounded-full bg-surface-container flex items-center justify-center text-on-surface-variant hover:text-on-surface transition-colors cursor-pointer"
   269→              title="Fermer"
   270→            >
   271→              <span className="material-symbols-outlined text-[18px]">close</span>
   272→            </button>
   273→          </div>
   274→        </div>
   275→
   276→        {/* Modal Body: Switch between Mode Player and Mode Browser (both use the same filtered iframe) */}
   277→        <div className="flex-1 overflow-y-auto flex flex-col bg-surface-container-low">
   278→          {viewMode === "player" ? (
   279→            /* ============================================================== */
   280→            /* MODE 1: STREAM PLAYER (Full Cinematic & TV Focus)               */
   281→            /* ============================================================== */
   282→            <div className="flex flex-col flex-1">
   283→              {/* Top Video Player Stage — filtered iframe from our proxy */}
   284→              <div className="relative aspect-video w-full bg-black flex items-center justify-center overflow-hidden group select-none max-h-[58vh]">
   285→                <iframe
   286→                  key={`${iframeKey}-${selectedServerIndex}-${selectedEpisodeIndex}`}
   287→                  src={currentIframeSrc}
   288→                  title={item.title}
   289→                  className="secure-iframe"
   290→                  referrerPolicy="no-referrer"
   291→                  sandbox="allow-scripts allow-same-origin allow-forms allow-presentation allow-popups"
   292→                  allow="autoplay; fullscreen; encrypted-media; picture-in-picture"
   293→                />
   294→
   295→                {/* Direct Stream Protection Watermark */}
   296→                <div className="absolute top-3 left-3 bg-black/70 backdrop-blur-md border border-primary/30 text-primary text-[11px] px-3 py-1 rounded-full flex items-center gap-1.5 shadow-md pointer-events-none z-10">
   297→                  <span className="material-symbols-outlined text-[14px]">shield</span>
   298→                  <span>
   299→                    Flux {item.platform} • Filtré par uBlock Origin (Zéro Pub)
   300→                  </span>
   301→                </div>
   302→
   303→                {/* Video Quality Badge */}
   304→                <div className="absolute top-3 right-3 bg-primary text-on-primary text-[11px] font-bold px-2 py-0.5 rounded shadow pointer-events-none z-10">
   305→                  {currentServer.quality}
   306→                </div>
   307→
   308→                {/* Big Center Play/Pause button */}
   309→                <button
   310→                  onClick={togglePlay}
   311→                  className="w-16 h-16 rounded-full bg-primary/90 text-on-primary flex items-center justify-center shadow-2xl transform active:scale-90 hover:scale-105 transition-all cursor-pointer z-20 opacity-0 group-hover:opacity-100 focus:opacity-100"
   312→                >
   313→                  <span className="material-symbols-outlined text-[36px] ml-0.5">
   314→                    {isPlaying ? "pause" : "play_arrow"}
   315→                  </span>
   316→                </button>
   317→
   318→                {/* Bottom video control overlay (decorative for cross-origin iframe) */}
   319→                <div className="absolute bottom-0 inset-x-0 p-3 bg-gradient-to-t from-black via-black/80 to-transparent flex flex-col gap-2 z-10 pointer-events-auto">
   320→                  {/* Timeline scrubber */}
   321→                  <div className="flex items-center gap-2">
   322→                    <span className="text-[11px] font-mono text-on-surface-variant">
   323→                      {formatSeconds(currentTime)}
   324→                    </span>
   325→                    <div
   326→                      className="flex-1 h-2 bg-white/20 rounded-full overflow-hidden cursor-pointer relative"
   327→                      onClick={(e) => {
   328→                        const rect = e.currentTarget.getBoundingClientRect();
   329→                        const pos = (e.clientX - rect.left) / rect.width;
   330→                        const newTime = pos * (duration || 180);
   331→                        setCurrentTime(newTime);
   332→                      }}
   333→                    >
   334→                      <div
   335→                        className="h-full bg-primary rounded-full transition-all"
   336→                        style={{ width: `${(currentTime / (duration || 180)) * 100}%` }}
   337→                      />
   338→                    </div>
   339→                    <span className="text-[11px] font-mono text-on-surface-variant">
   340→                      {formatSeconds(duration || 180)}
   341→                    </span>
   342→                  </div>
   343→
   344→                  {/* Player buttons */}
   345→                  <div className="flex items-center justify-between text-on-surface flex-wrap gap-2">
   346→                    <div className="flex items-center gap-3">
   347→                      <button
   348→                        onClick={togglePlay}
   349→                        className="hover:text-primary transition-colors cursor-pointer"
   350→                      >
   351→                        <span className="material-symbols-outlined text-[22px]">
   352→                          {isPlaying ? "pause" : "play_arrow"}
   353→                        </span>
   354→                      </button>
   355→
   356→                      {/* 10s backward/forward (decorative for iframe) */}
   357→                      <button
   358→                        onClick={() =>
   359→                          setCurrentTime((t) => Math.max(0, t - 10))
   360→                        }
   361→                        className="hover:text-primary transition-colors cursor-pointer"
   362→                        title="-10s"
   363→                      >
   364→                        <span className="material-symbols-outlined text-[20px]">replay_10</span>
   365→                      </button>
   366→                      <button
   367→                        onClick={() =>
   368→                          setCurrentTime((t) => Math.min(duration || 180, t + 10))
   369→                        }
   370→                        className="hover:text-primary transition-colors cursor-pointer"
   371→                        title="+10s"
   372→                      >
   373→                        <span className="material-symbols-outlined text-[20px]">forward_10</span>
   374→                      </button>
   375→
   376→                      {/* Volume Slider (decorative for iframe) */}
   377→                      <div className="flex items-center gap-1.5 ml-2">
   378→                        <button
   379→                          onClick={() => setIsMuted(!isMuted)}
   380→                          className="hover:text-primary transition-colors cursor-pointer"
   381→                        >
   382→                          <span className="material-symbols-outlined text-[20px]">
   383→                            {isMuted || volume === 0 ? "volume_off" : "volume_up"}
   384→                          </span>
   385→                        </button>
   386→                        <input
   387→                          type="range"
   388→                          min="0"
   389→                          max="100"
   390→                          value={isMuted ? 0 : volume}
   391→                          onChange={(e) => {
   392→                            setVolume(Number(e.target.value));
   393→                            setIsMuted(false);
   394→                          }}
   395→                          className="w-18 h-1 accent-primary bg-white/20 rounded-full cursor-pointer"
   396→                        />
   397→                      </div>
   398→                    </div>
   399→
   400→                    <div className="flex items-center gap-3">
   401→                      <span className="text-[11px] text-primary bg-primary/20 px-2 py-0.5 rounded font-bold">
   402→                        {currentServer.hoster} ({currentServer.language})
   403→                      </span>
   404→                      <button
   405→                        onClick={toggleFullscreen}
   406→                        className="hover:text-primary transition-colors cursor-pointer"
   407→                      >
   408→                        <span className="material-symbols-outlined text-[20px]">
   409→                          {isFullscreen ? "fullscreen_exit" : "fullscreen"}
   410→                        </span>
   411→                      </button>
   412→                    </div>
   413→                  </div>
   414→                </div>
   415→              </div>
   416→
   417→              {/* Streaming Controls Bar: Hosters & Episodes */}
   418→              <div className="p-4 sm:p-6 space-y-5 flex-1">
   419→                {/* Streaming Servers Selector (French-Stream Authentic Hosters) */}
   420→                <div className="bg-surface-container rounded-xl p-3.5 border border-outline-variant/15 flex flex-col md:flex-row md:items-center justify-between gap-3">
   421→                  <div className="flex items-center gap-2 flex-wrap">
   422→                    <span className="text-body-sm text-outline font-semibold">
   423→                      Serveurs de streaming :
   424→                    </span>
   425→                    <div className="flex gap-1.5 flex-wrap">
   426→                      {servers.map((srv, idx) => (
   427→                        <button
   428→                          key={srv.id}
   429→                          onClick={() => {
   430→                            setSelectedServerIndex(idx);
   431→                            setIsPlaying(true);
   432→                          }}
   433→                          className={`px-3 py-1.5 rounded-lg text-body-sm font-semibold transition-all cursor-pointer flex items-center gap-1.5 ${
   434→                            selectedServerIndex === idx
   435→                              ? "bg-primary text-on-primary font-bold shadow-sm"
   436→                              : "bg-surface-container-high text-on-surface-variant hover:text-on-surface"

   438→                        >
   439→                          <span>
   440→                            {srv.name} ({srv.hoster})
   441→                          </span>
   442→                          <span className="text-[10px] opacity-75 font-mono">[{srv.language}]</span>
   443→                        </button>
   444→                      ))}
   445→                    </div>
   446→                  </div>
   447→
   448→                  {/* Instant Direct Download Button */}
   449→                  <button
   450→                    onClick={() => onStartDownload(item)}
   451→                    className="px-4 py-2 rounded-xl bg-secondary text-on-secondary font-bold text-body-sm flex items-center gap-1.5 hover:scale-105 transition-all cursor-pointer shadow-md shrink-0"
   452→                  >
   453→                    <span className="material-symbols-outlined text-[18px]">download_for_offline</span>
   454→                    <span>
   455→                      Télécharger ({item.fileSizeMB ? `${(item.fileSizeMB / 1024).toFixed(1)} Go` : "HD"})
   456→                    </span>
   457→                  </button>
   458→                </div>
   459→
   460→                {/* Series Episode Selector if applicable */}
   461→                {episodes.length > 0 && (
   462→                  <div className="bg-surface-container rounded-xl p-3.5 border border-outline-variant/15 space-y-2">
   463→                    <div className="flex items-center justify-between">
   464→                      <span className="text-body-sm font-bold text-primary flex items-center gap-1.5">
   465→                        <span className="material-symbols-outlined text-[18px]">subscriptions</span>
   466→                        <span>Épisodes disponibles ({episodes.length})</span>
   467→                      </span>
   468→                      <span className="text-[11px] text-outline">Sans interruption publicitaire</span>
   469→                    </div>
   470→
   471→                    <div className="flex gap-2 overflow-x-auto pb-1 scrollbar-none">
   472→                      {episodes.map((ep, idx) => (
   473→                        <button
   474→                          key={ep.id}
   475→                          onClick={() => setSelectedEpisodeIndex(idx)}
   476→                          className={`px-3 py-2 rounded-lg text-[13px] font-semibold shrink-0 transition-all cursor-pointer text-left border ${
   477→                            selectedEpisodeIndex === idx
   478→                              ? "bg-primary text-on-primary border-primary font-bold shadow-sm"
   479→                              : "bg-surface-container-high border-outline-variant/15 text-on-surface-variant hover:text-on-surface"
   480→                          }`}
   481→                        >
   482→                          <div>Épisode {ep.episodeNumber}</div>
